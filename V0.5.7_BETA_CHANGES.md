```rust
pub async fn submit_mining_solution(
    State(state): State<Arc<AppState>>,
    Json(request): Json<MiningSolutionRequest>,
) -> Result<Json<ApiResponse<MiningSolutionResponse>>, StatusCode> {
    // ... existing validation ...

    // 🔐 AEGIS-KL AUTHENTICATION (v0.6.0)
    if let Some(miner_auth) = &state.miner_auth {
        // Extract signature and public key
        let signature = request.aegis_signature
            .ok_or_else(|| "AEGIS-KL signature required")?;
        let public_key_bytes = request.aegis_public_key
            .ok_or_else(|| "AEGIS-KL public key required")?;

        // Convert to proper types
        let aegis_public_key = q_aegis_ql::PublicKey::from_bytes(&public_key_bytes)
            .map_err(|_| "Invalid public key format")?;
        let aegis_sig = q_aegis_ql::Signature::from_bytes(&signature)
            .map_err(|_| "Invalid signature format")?;

        // Create solution data for signature verification
        let solution_data = format!("{}{}{}",
            request.hash,
            request.nonce,
            request.miner_address
        ).into_bytes();

        // Create credentials
        let credentials = q_mining::dev_fee::MinerCredentials {
            wallet_address: request.miner_address.clone(),
            aegis_public_key,
        };

        // Verify signature
        match miner_auth.verify_miner_auth(&credentials, &solution_data, &aegis_sig) {
            Ok(true) => {
                info!("✅ [AEGIS-KL] Miner {} authenticated", &request.miner_address[..16]);
            }
            Ok(false) => {
                return Ok(Json(ApiResponse::error(
                    "Invalid AEGIS-KL signature - unauthorized miner".to_string()
                )));
            }
            Err(e) => {
                warn!("❌ [AEGIS-KL] Verification error: {}", e);
                return Ok(Json(ApiResponse::error(
                    "Authentication failed".to_string()
                )));
            }
        }
    }

    // ... rest of existing code ...
}
```

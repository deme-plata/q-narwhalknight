```rust
// ========================================
// 🔐 AEGIS-KL AUTHENTICATION (v0.5.7+) - Post-Quantum Fork Protection
// ========================================
if let Some(miner_auth) = &state.miner_auth {
    if let (Some(sig_hex), Some(pk_hex)) = (&request.aegis_signature, &request.aegis_public_key) {
        // Decode hex strings
        let sig_bytes = hex::decode(sig_hex)?;
        let pk_bytes = hex::decode(pk_hex)?;

        // Convert to AEGIS-KL types
        let public_key = q_aegis_ql::PublicKey::from_bytes(&pk_bytes)?;
        let signature = q_aegis_ql::Signature::from_bytes(&sig_bytes)?;

        // Create solution data for verification
        let solution_data = format!("{}{}{}", hash, nonce, request.miner_address).into_bytes();

        // Create credentials
        let credentials = q_mining::dev_fee::MinerCredentials {
            wallet_address: request.miner_address.clone(),
            aegis_public_key: public_key,
        };

        // Verify signature
        match miner_auth.verify_miner_auth(&credentials, &solution_data, &signature) {
            Ok(true) => {
                info!("✅ [AEGIS-KL] Miner {} authenticated", &request.miner_address[..16]);
            }
            Ok(false) => {
                return Ok(Json(ApiResponse::error(
                    "Invalid AEGIS-KL signature - mining submission rejected".to_string()
                )));
            }
            Err(e) => {
                return Ok(Json(ApiResponse::error(
                    "AEGIS-KL authentication failed".to_string()
                )));
            }
        }
    } else {
        // Signature REQUIRED when authentication is enabled
        return Ok(Json(ApiResponse::error(
            "AEGIS-KL signature required for mining submissions (upgrade your miner software)".to_string()
        )));
    }
}
```

```rust
pub async fn sync_to_height(&self, target_height: u64) -> Result<()> {
    let mut received_packs = 0;
    let total_chunks = chunks.len();

    for chunk in chunks {
        // Create response channel
        let (response_tx, response_rx) = oneshot::channel();
        let request_id = generate_unique_id();

        // ✅ Register response channel in shared map
        self.response_map.lock().await.insert(
            request_id.clone(),
            response_tx
        );

        // Send request
        send_block_pack_request(start, end, request_id).await?;

        // ✅ WAIT for response with 30s timeout
        match timeout(Duration::from_secs(30), response_rx).await {
            Ok(Ok(pack)) => {
                info!("✅ [TURBO SYNC] Received pack {}-{}",
                      pack.start_height, pack.end_height);

                // Apply pack to storage
                self.apply_block_pack(pack, None).await?;
                received_packs += 1;
            }
            Ok(Err(e)) => {
                error!("❌ [TURBO SYNC] Pack response error: {}", e);
                anyhow::bail!("Failed to receive pack: {}", e);
            }
            Err(_) => {
                error!("❌ [TURBO SYNC] Timeout waiting for pack response");
                anyhow::bail!("Timeout after 30s waiting for block pack");
            }
        }
    }

    // ✅ Verify ALL packs received
    if received_packs != total_chunks {
        anyhow::bail!(
            "Incomplete sync: received {}/{} packs",
            received_packs, total_chunks
        );
    }

    // ✅ Verify blocks written to storage
    let new_height = self.get_latest_qblock_height().await?
        .unwrap_or(0);

    if new_height < target_height {
        anyhow::bail!(
            "TURBO SYNC INCOMPLETE: Expected height {}, got {}",
            target_height, new_height
        );
    }

    // ✅ Only return Ok() after verification
    info!("✅ [TURBO SYNC] Successfully synced to height {}",
          new_height);
    Ok(())
}
```

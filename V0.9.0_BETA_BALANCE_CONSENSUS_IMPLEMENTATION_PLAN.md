```rust
/// Calculate balance updates for mining solutions
/// Returns deterministic balance updates to be included in block
async fn calculate_balance_updates(
    &self,
    solutions: &[MiningSolution],
    wallet_balances: &Arc<RwLock<HashMap<Address, u64>>>,
) -> Vec<BalanceUpdate> {
    use chrono::Utc;
    use crate::handlers::{calculate_block_reward_time_based, GENESIS_TIMESTAMP};

    let mut balance_updates = Vec::new();
    let balances = wallet_balances.read().await;
    let current_timestamp = Utc::now().timestamp() as u64;
    let block_reward_total = calculate_block_reward_time_based(GENESIS_TIMESTAMP, current_timestamp);

    // Apply 1% development fee
    const DEV_FEE_PERCENT: f64 = 0.01;
    let dev_fee_amount = (block_reward_total as f64 * DEV_FEE_PERCENT) as u64;
    let miner_reward = block_reward_total - dev_fee_amount;

    // Group solutions by miner address
    let mut miner_solutions: HashMap<Address, usize> = HashMap::new();
    for solution in solutions {
        *miner_solutions.entry(solution.miner_address).or_insert(0) += 1;
    }

    // Create balance updates for each miner
    for (miner_address, solution_count) in miner_solutions {
        let old_balance = balances.get(&miner_address).copied().unwrap_or(0);
        let reward = miner_reward * solution_count as u64;
        let new_balance = old_balance + reward;

        balance_updates.push(BalanceUpdate {
            address: miner_address,
            old_balance,
            new_balance,
            reason: format!("mining_reward_{}_solutions", solution_count),
            timestamp: current_timestamp,
        });
    }

    // Add development fee balance update
    if dev_fee_amount > 0 {
        const FOUNDER_WALLET_HEX: &str = "efca1e8c1f46e91013b4073898c771bb3d566453537ccf87e834505925e50723";
        let dev_wallet: Address = hex::decode(FOUNDER_WALLET_HEX)
            .unwrap()
            .try_into()
            .unwrap();

        let old_dev_balance = balances.get(&dev_wallet).copied().unwrap_or(0);
        let new_dev_balance = old_dev_balance + dev_fee_amount;

        balance_updates.push(BalanceUpdate {
            address: dev_wallet,
            old_balance: old_dev_balance,
            new_balance: new_dev_balance,
            reason: "dev_fee".to_string(),
            timestamp: current_timestamp,
        });
    }

    balance_updates
}
```

### Step 3: Update Block Processor (`main.rs`)

**Location**: Find where blocks are received via P2P or turbo_sync

**Required Logic**:
```rust
/// Apply balance updates from a received block
/// CRITICAL: This ensures balance consensus across all nodes
async fn apply_balance_updates_from_block(
    block: &QBlock,
    wallet_balances: &Arc<RwLock<HashMap<Address, u64>>>,
    storage: &Storage,
) -> anyhow::Result<()> {
    if block.balance_updates.is_empty() {
        // Old block format (v0.8.x) - skip balance updates
        return Ok(());
    }

    info!("📊 Applying {} balance updates from block height {}",
          block.balance_updates.len(), block.header.height);

    let mut balances = wallet_balances.write().await;

    for update in &block.balance_updates {
        let current_balance = balances.get(&update.address).copied().unwrap_or(0);

        // VERIFICATION: Check if current balance matches expected old_balance
        if current_balance != update.old_balance {
            warn!("⚠️  Balance verification failed for {}: expected {}, got {}. Forcing update.",
                  hex::encode(&update.address[..8]),
                  update.old_balance,
                  current_balance);
        }

        // Apply balance update
        balances.insert(update.address, update.new_balance);

        // Persist to RocksDB
        storage.set_balance(&update.address, update.new_balance).await?;

        info!("💰 Balance updated: {} -> {} ({})",
              hex::encode(&update.address[..8]),
              update.old_balance as f64 / 100_000_000.0,
              update.new_balance as f64 / 100_000_000.0);
    }

    Ok(())
}
```

## ⚠️ Risks & Considerations

### Risk 1: Breaking Change

**Issue**: Requires ALL nodes to upgrade
**Mitigation**:
- Use `#[serde(default)]` for backwards compatibility
- Deploy gradually (testnet first)
- Announce upgrade window

### Risk 2: Balance Forks

**Issue**: If two nodes have different balance states before upgrade
**Mitigation**:
- Reset testnet balances at upgrade
- Provide migration tool
- Clear documentation

### Risk 3: Increased Block Size

**Issue**: balance_updates adds data to every block
**Mitigation**:
- Compress balance_updates (if >100 updates)
- Monitor block propagation latency
- Optimize serialization

### Risk 4: Database Migration

**Issue**: Existing blocks don't have balance_updates
**Mitigation**:
- Old blocks work fine (empty vec default)
- Only new blocks include balance_updates
- No migration needed for blockchain data

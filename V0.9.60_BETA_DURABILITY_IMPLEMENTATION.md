**File:** `crates/q-storage/src/kv.rs` (lines 833-865)

## 📚 USAGE EXAMPLES

### Automated Hourly Backups

```rust
use chrono::Utc;
use tokio::time::{interval, Duration};

async fn backup_loop(storage: Arc<RocksDBKV>) {
    let mut ticker = interval(Duration::from_secs(3600)); // 1 hour

    loop {
        ticker.tick().await;

        let timestamp = Utc::now().format("%Y-%m-%d-%H-%M").to_string();
        let backup_path = format!("./backups/checkpoint-{}", timestamp);

        // Create checkpoint
        if let Err(e) = storage.create_checkpoint(&backup_path).await {
            error!("❌ [BACKUP] Failed: {}", e);
            continue;
        }

        // Verify integrity
        match storage.verify_checkpoint(&backup_path).await {
            Ok(true) => info!("✅ [BACKUP] Verified: {}", backup_path),
            Ok(false) => {
                error!("❌ [BACKUP] CORRUPT! Deleting.");
                let _ = std::fs::remove_dir_all(&backup_path);
            }
            Err(e) => error!("❌ [BACKUP] Verification error: {}", e),
        }

        // Keep only last 24 backups
        cleanup_old_backups("./backups", 24).await;
    }
}
```

### Recovery from Corruption

```bash
# 1. Stop node
systemctl stop q-api-server

# 2. List checkpoints
ls -lht ./backups/

# 3. Backup corrupted DB
mv ./data-mine6 ./data-mine6-CORRUPTED

# 4. Restore (instant - hard links!)
cp -al ./backups/checkpoint-2025-11-08-12-00 ./data-mine6

# 5. Restart
systemctl start q-api-server

# Recovery time: <1 minute (vs hours for full reindex!)
```

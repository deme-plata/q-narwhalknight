### Step 2: Add Mining Heartbeat Monitor Task

In `/opt/orobit/shared/q-narwhalknight/crates/q-api-server/src/main.rs` after line 3187 (before time-based block production loop):

```rust
// ========================================
// 💓 MINING HEARTBEAT MONITOR (v0.8.9-beta)
// Detects when mining stops and alerts operators
// ========================================
{
    let app_state_heartbeat = app_state.clone();
    tokio::spawn(async move {
        info!("💓 Starting mining heartbeat monitor (v0.8.9-beta)...");
        let mut interval = tokio::time::interval(tokio::time::Duration::from_secs(60));
        let mut alert_sent = false;

        loop {
            interval.tick().await;

            let last_solution_time = app_state_heartbeat
                .last_mining_solution_time
                .load(std::sync::atomic::Ordering::SeqCst);

            let current_time = chrono::Utc::now().timestamp() as u64;
            let time_since_last_solution = current_time.saturating_sub(last_solution_time);

            const STALL_THRESHOLD_SECS: u64 = 300; // 5 minutes

            if time_since_last_solution > STALL_THRESHOLD_SECS {
                // Mining has stalled!
                app_state_heartbeat.mining_is_healthy.store(false, std::sync::atomic::Ordering::SeqCst);

                if !alert_sent {
                    error!("🚨 MINING STALL DETECTED!");
                    error!("   No solutions received for {} seconds", time_since_last_solution);
                    error!("   Last solution: {} seconds ago", time_since_last_solution);
                    error!("   ACTION REQUIRED: Check miner status or restart miners");

                    // Broadcast SSE alert to frontend
                    let _ = app_state_heartbeat.event_broadcaster.broadcast(
                        q_api_server::streaming::StreamEvent::Custom {
                            event_type: "mining_stall_detected".to_string(),
                            data: serde_json::json!({
                                "time_since_last_solution": time_since_last_solution,
                                "threshold_seconds": STALL_THRESHOLD_SECS,
                                "recommended_action": "Restart miners or check miner logs",
                                "timestamp": chrono::Utc::now().to_rfc3339(),
                            }).to_string(),
                        }
                    );

                    alert_sent = true;
                }
            } else {
                // Mining is healthy
                if !app_state_heartbeat.mining_is_healthy.load(std::sync::atomic::Ordering::SeqCst) {
                    info!("✅ Mining recovered! Solutions arriving normally.");
                    app_state_heartbeat.mining_is_healthy.store(true, std::sync::atomic::Ordering::SeqCst);

                    let _ = app_state_heartbeat.event_broadcaster.broadcast(
                        q_api_server::streaming::StreamEvent::Custom {
                            event_type: "mining_recovered".to_string(),
                            data: serde_json::json!({
                                "recovery_time": chrono::Utc::now().to_rfc3339(),
                            }).to_string(),
                        }
                    );

                    alert_sent = false;
                }
            }
        }
    });
}
```

### Step 3: Add Mining Health API Endpoint

In `/opt/orobit/shared/q-narwhalknight/crates/q-api-server/src/handlers.rs`:

```rust
/// Mining health status response
#[derive(serde::Serialize)]
pub struct MiningHealthResponse {
    pub is_healthy: bool,
    pub time_since_last_solution: u64, // seconds
    pub last_solution_timestamp: u64,   // Unix timestamp
    pub status: String,                 // "healthy" or "stalled"
    pub threshold_seconds: u64,         // Stall detection threshold
}

/// GET /api/v1/mining/health - Check if mining is active
pub async fn get_mining_health(
    State(app_state): State<Arc<AppState>>,
) -> Result<Json<MiningHealthResponse>, (StatusCode, String)> {
    let last_solution_time = app_state.last_mining_solution_time.load(std::sync::atomic::Ordering::SeqCst);
    let current_time = chrono::Utc::now().timestamp() as u64;
    let time_since_last_solution = current_time.saturating_sub(last_solution_time);
    let is_healthy = app_state.mining_is_healthy.load(std::sync::atomic::Ordering::SeqCst);

    const STALL_THRESHOLD: u64 = 300; // 5 minutes

    Ok(Json(MiningHealthResponse {
        is_healthy,
        time_since_last_solution,
        last_solution_timestamp: last_solution_time,
        status: if is_healthy { "healthy".to_string() } else { "stalled".to_string() },
        threshold_seconds: STALL_THRESHOLD,
    }))
}
```

Then register the route in `main.rs`:

```rust
.route("/api/v1/mining/health", get(q_api_server::handlers::get_mining_health))
```

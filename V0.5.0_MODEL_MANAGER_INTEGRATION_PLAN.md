```rust
/// Request to switch model
#[derive(Debug, Deserialize)]
pub struct SwitchModelRequest {
    pub model: String, // e.g., "Mistral-Small-3.2-24B-Instruct"
}

/// Response for model switch
#[derive(Debug, Serialize)]
pub struct SwitchModelResponse {
    pub success: bool,
    pub model_info: ModelInfo,
}

#[derive(Debug, Serialize)]
pub struct ModelInfo {
    pub name: String,
    pub parameters_billions: f32,
    pub layers: usize,
    pub quantization: String,
    pub ram_usage_mb: usize,
}

/// POST /api/v1/chat/:id/switch-model - Switch chat to different model
pub async fn switch_model(
    State(state): State<Arc<AppState>>,
    Path(chat_id): Path<String>,
    Json(req): Json<SwitchModelRequest>,
) -> Result<Json<ApiResponse<SwitchModelResponse>>, StatusCode> {
    info!("🔄 Switching chat {} to model: {}", chat_id, req.model);

    // Validate model exists and is supported
    let metadata = q_ai_inference::ModelMetadata::from_name(
        &req.model,
        "http://quillon.xyz"
    ).map_err(|e| {
        error!("Invalid model name {}: {}", req.model, e);
        StatusCode::BAD_REQUEST
    })?;

    // Update chat metadata
    let mut chat_metadata = state.storage_engine
        .get_chat_metadata(&chat_id)
        .await
        .map_err(|_| StatusCode::NOT_FOUND)?;

    chat_metadata.model = req.model.clone();
    chat_metadata.updated_at = current_timestamp();

    state.storage_engine
        .update_chat_metadata(&chat_metadata)
        .await
        .map_err(|e| {
            error!("Failed to update chat metadata: {}", e);
            StatusCode::INTERNAL_SERVER_ERROR
        })?;

    // Optionally pre-load the model
    if let Some(manager) = &state.ai_model_manager {
        info!("⏳ Pre-loading model: {}", req.model);
        let _ = manager.get_or_load_model(&req.model).await.map_err(|e| {
            warn!("Failed to pre-load model (will load on next message): {}", e);
        });
        info!("✅ Model switched and loaded");
    }

    Ok(Json(ApiResponse::success(SwitchModelResponse {
        success: true,
        model_info: ModelInfo {
            name: metadata.name.clone(),
            parameters_billions: metadata.parameters_billions,
            layers: metadata.layer_count,
            quantization: "Q4_K_M".to_string(),
            ram_usage_mb: metadata.ram_usage_mb,
        },
    })))
}
```

```rust
//! Chat templates for different AI models
//!
//! Each model has a specific prompt format that must be followed for optimal results.

/// Format prompt for Mistral-7B-Instruct-v0.3
pub fn format_mistral_7b(system_prompt: Option<&str>, user_message: &str) -> String {
    if let Some(system) = system_prompt {
        format!("[INST] {} [/INST]\n[INST] {} [/INST]", system, user_message)
    } else {
        format!("[INST] {} [/INST]", user_message)
    }
}

/// Format prompt for Mistral-Small-3.2-24B-Instruct
///
/// Template format: `[SYSTEM_PROMPT]{system}[/SYSTEM_PROMPT][INST]{user}[/INST]{assistant}`
pub fn format_mistral_small_3_2(system_prompt: Option<&str>, user_message: &str) -> String {
    if let Some(system) = system_prompt {
        format!(
            "[SYSTEM_PROMPT]{}[/SYSTEM_PROMPT][INST]{}[/INST]",
            system, user_message
        )
    } else {
        format!("[INST]{}[/INST]", user_message)
    }
}

/// Format prompt for Llama models
pub fn format_llama(system_prompt: Option<&str>, user_message: &str) -> String {
    if let Some(system) = system_prompt {
        format!(
            "<s>[INST] <<SYS>>\n{}\n<</SYS>>\n\n{} [/INST]",
            system, user_message
        )
    } else {
        format!("<s>[INST] {} [/INST]", user_message)
    }
}

/// Auto-detect and format prompt based on model name
pub fn format_prompt(model_name: &str, system_prompt: Option<&str>, user_message: &str) -> String {
    if model_name.contains("Mistral-Small-3.2") {
        format_mistral_small_3_2(system_prompt, user_message)
    } else if model_name.contains("Mistral-7B") || model_name.contains("Mistral-") {
        format_mistral_7b(system_prompt, user_message)
    } else if model_name.contains("Llama") {
        format_llama(system_prompt, user_message)
    } else {
        // Default: Mistral format
        format_mistral_7b(system_prompt, user_message)
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_mistral_small_format() {
        let prompt = format_mistral_small_3_2(
            Some("You are a helpful AI assistant."),
            "Hello, how are you?"
        );
        assert_eq!(
            prompt,
            "[SYSTEM_PROMPT]You are a helpful AI assistant.[/SYSTEM_PROMPT][INST]Hello, how are you?[/INST]"
        );
    }

    #[test]
    fn test_mistral_7b_format() {
        let prompt = format_mistral_7b(None, "Explain quantum computing");
        assert_eq!(prompt, "[INST] Explain quantum computing [/INST]");
    }

    #[test]
    fn test_auto_detect() {
        let prompt1 = format_prompt("Mistral-Small-3.2-24B-Instruct", None, "Hello");
        assert!(prompt1.starts_with("[INST]Hello[/INST]"));

        let prompt2 = format_prompt("Mistral-7B-Instruct-v0.3", None, "Hello");
        assert_eq!(prompt2, "[INST] Hello [/INST]");
    }
}
```

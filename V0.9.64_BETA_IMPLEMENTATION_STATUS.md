    match storage_fallback.get_highest_contiguous_block().await {
        Ok(current) if current < end_height => {
            warn!("🔄 [P2P GAP-FILL] Gossipsub timed out, using direct peer requests");

            // Get connected peers
            let peers = /* need network_manager reference */;

            if peers.is_empty() {
                error!("❌ [P2P GAP-FILL] No peers connected");
                return;
            }

            // Request blocks from multiple peers in parallel
            let blocks_needed = end_height - current;
            let blocks_per_peer = (blocks_needed / peers.len() as u64).max(1);

            for (idx, peer_id) in peers.iter().enumerate() {
                let start = current + 1 + (idx as u64 * blocks_per_peer);
                let end = (start + blocks_per_peer - 1).min(end_height);

                // Send BlockSyncRequest to each peer
                network_tx.send(NetworkCommand::RequestBlocks {
                    peer_id: peer_id.clone(),
                    start_height: start,
                    limit: (end - start + 1) as usize,
                }).await;
            }
        }
    }
});
